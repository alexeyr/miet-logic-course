\begin{frame}
    \frametitle{Структура атомарных высказываний}
    \begin{itemize}
        \item Вспомним примеры простых высказываний с прошлой лекции: $4<5$, \enquote{Волга впадает в Балтийское море}.
        \pause
        \item $4$, $5$, \enquote{Волга} и \enquote{Балтийское море} здесь \emph{константы}, обозначающие какие-то объекты.
        \item $<$ и \enquote{\_ впадает в \_} "--- \emph{предикаты}, обозначающие отношения между объектами. Оба предиката \emph{бинарные} (или \emph{двухместные}).
        \pause
        \item Свойства (например \enquote{\_ круглое} в \enquote{Земля круглая}) "--- тоже предикаты, но \emph{унарные}.
        \pause
        \item Ещё пример: $2+2=5$. Здесь $2$ и $5$ "--- \pause константы, $=$ "--- \pause предикат, а $+$ "--- \pause \emph{функция}.
        \item $2+2$ "--- \emph{терм}, обозначает объект (как и константы).
        \item Применяя $n$-местный предикат к $n$ термам, получаем атомарную формулу.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Сигнатуры и термы}
    \begin{itemize}
        \item \emph{Сигнатура} $\Sigma$ задаёт конечные или счётные множества символов: констант $\Const_\Sigma$, функций $\Fun_\Sigma$ и предикатов $\Pred_\Sigma$. Для функций и предикатов задана \emph{арность} (число аргументов).
        \item \emph{Переменные} "--- $x,\,y,\,z_3,\,\ldots$. Обозначают \pause какие-то объекты (не истину/ложь, как $p,q,r$). Множество $\Var$ не зависит от сигнатуры. \pause
        \item \emph{Термы} обозначают объекты. Множество $\Term_\Sigma$ задаётся по индукции:
        \begin{itemize}
            \item переменные и константные символы "--- термы;
            \item если $f$ имеет арность $n$, а $t_1,\ldots,t_n$ "--- термы, то $f(t_1,\ldots,t_n)$ "--- терм;
            \item других термов нет.
        \end{itemize}
        \pause
        \item Примеры: $x$, $(y+2)\cdot z$ (в подходящей сигнатуре).
        \item \emph{Замкнутые термы} не содержат переменных. Например, $2+2$ и $\pi$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Формулы и область действия кванторов}
    \begin{itemize}
        \item Множество формул $\Frm_\Sigma$ задаётся по индукции:
        \begin{itemize}
            \item если $P$ имеет арность $n$, а $t_1,\ldots,t_n$ "--- термы, то $P(t_1,\ldots,t_n)$ "--- атомарная формула;
            \item если $A,B$ "--- формулы, то $\neg A$, $(A\land B)$, $(A\lor B)$, $(A\to B)$ "--- формулы;
            \item если $A$ "--- формула, а $x$ "--- переменная, то $\forall x A$ и $\exists x A$ "--- формулы;
            \item других формул нет.
        \end{itemize}
        \pause
        \item В $\forall x A$ и $\exists x A$ формула $A$ "--- \emph{область действия} квантора.
        \item Договоримся: без скобок квантор действует на всю формулу справа до конца текущих скобок.
        \pause
        \item $\forall x\,P(x)\to Q(x)$ означает $\forall x\,(P(x)\to Q(x))$, а не $(\forall x\,P(x))\to Q(x)$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Модели}
    \begin{itemize}
        \item \emph{Модель} $M$ сигнатуры $\Sigma$ состоит из:
        \begin{itemize}
            \item \emph{Носитель} (или \emph{универсум}): \pause непустое множество $\bar{M}$.
            \item Для каждого символа $c : \Const_\Sigma$\only<2->{\footnote{Я пишу $:$ вместо $\in$ для объявления новой переменной.}}: \pause элемент $c_M : \bar{M}$.
            \item Для $f: \Fun_\Sigma$: \pause функция $f_M: \bar{M}^{\arity(f)} \to \bar{M}$.
            \item Для $P: \Pred_\Sigma$: \pause предикат $P_M: \bar{M}^{\arity(P)} \to \{0,1\}$.
        \end{itemize}
        \item \emph{Оценка} $\sigma : \Var \to \bar{M}$ задаёт значение каждой переменной в модели.
        \item Важно: символы сигнатуры сами по себе ничего не говорят об их интерпретации! $+_M$ может быть $-$ или $\cdot$ или $x,y \mapsto \int_{x}^{y} t\,dt$.
        \pause 
        \item Одно исключение: если есть $=$, то $=_M$ это равенство на $\bar{M}$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Значения термов и формул}
    \begin{itemize}
        \item $\sigma$ расширяется на \only<1>{$\Term_\Sigma \to ~ ???$ и на $\Frm_\Sigma \to ~ ???$}\only<2->{$\Term_\Sigma \to \bar{M}$ и на $\Frm_\Sigma \to \{0,1\}$} по индукции: \pause 
        \item $\sigma(c) = c_M$.
        \item $\sigma(f(t_1,\ldots,t_n)) = f_M(\sigma(t_1),\ldots,\sigma(t_n))$.
        \pause
        \item $\sigma(P(t_1,\ldots,t_n)) = P_M(\sigma(t_1),\ldots,\sigma(t_n))$.
        \item $\sigma(A \land B) = \sigma(A) \land \sigma(B)$ (аналогично для $\neg$/$\lor$/$\to$).
        \pause
        \item Для переменной $v$ и объекта $a:\bar M$ определим:
        \[ \begin{array}{l}
            \sigma_{v \mapsto a}(v) = a \\
            \sigma_{v \mapsto a}(u) = \sigma(u) \text{ для остальных переменных } u.
        \end{array} \]
        \pause
        Теперь
        \[ \sigma(\forall v ~ A) = \left\{ \begin{array}{l}
            1, \text{ если } \forall a : \bar{M} ~  \sigma_{v \mapsto a}(A) = 1 \\
            0 \text{ иначе }
        \end{array} \right. \]
        и аналогично для $\exists$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Свободные и связанные переменные}
    \begin{itemize}
        \item \emph{Связанное вхождение переменной} "--- \pause $v$ под квантором по этой же переменной $\forall v/\exists v$.
        \item \emph{Свободное вхождение переменной} "--- \pause любое другое.
        \item Переменная свободна в формуле, если имеет какие-то свободные вхождения.
        \item \emph{Замкнутая формула} (или \emph{предложение}) "--- формула без свободных переменных.
        \item Почему это важно? Какая между ними разница? \pause 
        \item $\sigma(\forall/\exists v ~ \ldots)$ не зависит от $\sigma(v)$.
        \pause
        \item Отсюда можно доказать, что значение формулы $A$ зависит только от значений свободных переменных $A$, но не от связанных.
        \pause
        \item А значение замкнутой формулы? \pause Вообще не зависит от оценки, только от модели.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Свободные и связанные переменные (2)}
    \begin{itemize}
        \item Связанную переменную можно переименовать в свежую (не встречающуюся в формуле), сохранив смысл.
        \item Вместо свободной переменной можно подставить произвольный терм, вместо связанной нельзя.
        \item Переменные с одним названием, связанные разными кванторами, это по сути разные переменные. Пример: $x = 3 \land (\exists x ~ x=0) \land \forall x ~ (x > 0 \to x^2 > 0)$.
        \item В математике переменные могут связываться не только кванторами: \pause
        \[ \lim_{x \to \ldots} \ldots \qquad \int_{\ldots}^{\ldots} \ldots dx \qquad \{x|\ldots\} \]
        все связывают $x$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Примеры}
    \begin{itemize}
        \item Формула (замкнутая): $\forall x ~ (P(x) \to Q(x))$.
        \item Возьмём модель с носителем $\bar{M} = \{0,1,2\}$.
        \item Как можно задать предикаты $P$ и $Q$? \pause
        \item[] \begin{tabular}{c | c c c}
            x & 0 & 1 & 2 \\ \hline
            P(x) & 0 & 0 & 1 \\
            Q(x) & 1 & 0 & 1
        \end{tabular}
        \item Истинна ли формула? \pause Да. \pause
        \item Попробуем другую формулу на той же модели:  $(\exists x ~ P(x) \land Q(x)) \land \exists x ~ \neg P(x)$. Истинна ли она? \pause Да. \pause
        \item Как насчёт незамкнутой формулы $P(x) \to \forall x ~ Q(x)$? \pause Истинна при $\sigma(x) = 0,~\sigma(x) = 1$, ложна при $\sigma(x) = 2$. \pause
        \item Для простоты можно писать $x = 0$ вместо $\sigma(x) = 0$. \pause
        \item Переименование объектов $0,1,2$ в $a,b,c$ не изменит ответов.
    \end{itemize}
\end{frame}

\begin{frame}
   \frametitle{Формализация}
   \begin{itemize}
       \item Часто возникает задача: дано утверждение (математическое или в терминах \enquote{реального мира}), нужно записать его в виде формулы данной сигнатуры.
       \item Пример: \enquote{Кто-то любит всех на свете}. Универсум: люди, предикат $\ident{Loves}(x, y)$ для \enquote{$x$ любит $y$}.
       \item Кто-то любит всех на свете $\equiv \exists x $ $x$ любит всех на свете \pause $\equiv \exists x ~ \forall y $ $x$ любит $y$ \pause $\equiv \exists x ~ \forall y ~ \ident{Loves}(x, y)$.
   \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Многосортная логика предикатов}
    \begin{itemize}
        \item Иногда нужны разные типы объектов: например, числа, множества чисел и функции.
        \item К сигнатуре добавляются \emph{сорта}. Каждая переменная и константа имеет сорт: $x:S$, $c:S$.
        \item У функций и предикатов заданы сорта аргументов, у функций "--- ещё и сорт результата.
        \item Аргументы должны иметь нужные сорта; иначе запись не является термом или формулой.
        \item В модели каждому сорту соответствует непустой носитель.
        \item Многосортную логику можно свести к односортной: сорта задаются предикатами, а кванторы ограничиваются ими. Нужны также условия на интерпретации символов.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Подстановки}
    \begin{itemize}
        \item Вернёмся к односортной логике. Заменим переменную $v$ на терм $t$ в терме $s$ и формуле $A$: $s[v \mapsto t]$ и $A[v \mapsto t]$. Должно получиться
        \[ \begin{array}{c}
            \sigma(s[v \mapsto t]) = \sigma_{v \mapsto \sigma(t)}(s) \\
            \sigma(A[v \mapsto t]) = \sigma_{v \mapsto \sigma(t)}(A)
        \end{array} \]
        (понятен ли смысл этого?). 
        \pause
        \item Первые шаги очевидны:
        \[ \begin{array}{l}
            v[v \mapsto t] = t \\ \pause
            u[v \mapsto t] = \pause u ~(u\text{ "--- переменная, кроме }v) \\
            c[v \mapsto t] = \pause c ~(c \text{ "--- константа}) \\ \pause
            f(t_1,\ldots,t_n)[v \mapsto t] = \pause f(t_1[v \mapsto t],\ldots,t_n[v \mapsto t]) \\ \pause
            P(t_1,\ldots,t_n)[v \mapsto t] = \pause P(t_1[v \mapsto t],\ldots,t_n[v \mapsto t]) \\ \pause
            (\neg A)[v \mapsto t] = \neg (A[v \mapsto t]) ~ (\text{аналогично для } \land/\lor/\to)
        \end{array} \]
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Подстановки с кванторами}
    \begin{itemize}
        % \pause внутри \[...\] или сразу после \] даёт лишнюю пустую строку (см. \reveal в преамбуле).
        \item Для $\forall$ (для $\exists$ аналогично):
        \[ (\forall v ~ A)[v \mapsto t] = \reveal<+(1)->{\forall v ~ A} \]
        \item<+(1)-> При $u\ne v$ можно предположить
        \[ (\forall u ~ A)[v \mapsto t] = \reveal<+(1)->{\forall u ~ (A[v \mapsto t])} \]
        \item<+(1)-> Формула $\exists y~x=2\cdot y$ означает «$x$ делится на $2$». Заменим $x$ на $y$: хотим получить \pause \enquote{$y$ делится на $2$}.
        \item Но получаем \pause $\exists y~y=2\cdot y$: квантор связал $y$!
        \pause
        \item Сначала переименуем связанные переменные, мешающие подстановке, в свежие для $A$ и $t$:
        \[ (\exists y~x=2\cdot y)[x\mapsto y]
             = (\exists y'~x=2\cdot y')[x\mapsto y]
             = \exists y'~y=2\cdot y'. \]
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Итоги}
    \begin{itemize}
        \item Сигнатура задаёт символы языка; из них строятся термы и формулы.
        \item Модель задаёт объекты и интерпретации символов, оценка "--- значения переменных.
        \item Термы обозначают объекты, формулы принимают значения истины и лжи.
        \item Значение формулы зависит только от оценки её свободных переменных; значение замкнутой формулы "--- только от модели.
        \item При формализации следим за универсумом, областью действия кванторов и свободными переменными.
        \item Подстановка заменяет свободные вхождения переменной и не должна делать переменные терма связанными.
    \end{itemize}
\end{frame}
