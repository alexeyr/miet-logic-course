\begin{frame}
    \frametitle{Правила для логики высказываний}
    \scriptsize
    \setlength{\abovedisplayskip}{2pt}
    \setlength{\belowdisplayskip}{2pt}
    \setlength{\abovedisplayshortskip}{0pt}
    \setlength{\belowdisplayshortskip}{2pt}
    \begin{itemize}
        \item Для $\land$ ($l$ "--- левый операнд, $r$ "--- правый):
        \[
        \frac{A\qquad B}{A\land B}\ \land I
        \qquad\qquad
        \frac{A\land B}{A}\ \land E_l
        \qquad
        \frac{A\land B}{B}\ \land E_r
        \]
        \item Для $\to$:
        \[
        \frac{\ndsubproof{A}{B}}{A\to B}\ \to I
        \qquad\qquad
        \frac{A\to B\quad A}{B}\ \to E
        \]
        \item Для $\neg$ и $\bot$:
        \[
        \frac{\ndsubproof{A}{\bot}}{\neg A}\ \neg I
        \qquad\qquad
        \frac{\neg A\quad A}{\bot}\ \neg E/\bot I
        \qquad\qquad
        \frac{\bot}{A}\ \bot E
        \]
        \item Для $\lor$:
        \[
        \frac{A}{A\lor B}\ \lor I_l
        \qquad
        \frac{B}{A\lor B}\ \lor I_r
        \qquad\qquad
        \frac{A\lor B\quad\ndsubproof{A}{C}\quad\ndsubproof{B}{C}}{C}\ \lor E
        \]
        \item Остальные:
        \[
        \frac{A}{A}\ R
        \qquad\qquad
        \frac{\ndsubproof{\neg A}{\bot}}{A}\ \RAA
        \qquad\qquad
        \frac{\ndsubproof{\neg B}{A}}{A\lor B}\ \lor I'_l
        \qquad
        \frac{\ndsubproof{\neg A}{B}}{A\lor B}\ \lor I'_r
        \]
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Правила для кванторов}
    \begin{itemize}
        \item Все правила логики высказываний сохраняются.
        \item Добавляются правила введения и исключения для кванторов. \pause
        \item Простые правила:
        \[
            \frac{\forall x~A(x)}{A(t)}\ \forall E
            \qquad\qquad
            \frac{A(t)}{\exists x~A(x)}\ \exists I.
        \]
        \item $t$ "--- терм без свободных переменных, например параметр.
        \item Можно использовать уже имеющиеся параметры. \pause
        \item Правила с новым параметром:
        \[
            \frac{\ndsubproof{\text{произв. }a}{A(a)}}{\forall x~A(x)}\ \forall I
            \qquad
            \frac{\exists x~A(x)\quad\ndsubproof{a\text{ такой, что }A(a)}{B}}{B}\ \exists E.
        \]
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Пояснения к правилам}
    \begin{itemize}
        \item $\forall I$: доказываем $A(a)$ для произвольного $a$, затем заключаем $\forall x~A(x)$.
        \item $a$ не должно встречаться в оставшихся посылках и допущениях или в заключении правила. \pause
        \item $\exists E$: вводим свежий $a$ и временно допускаем $A(a)$. Из него выводим $B$ и снимаем это допущение.
        \item $a$ не должно встречаться в $B$ или в других доступных посылках и допущениях. \pause
        \item Удобно брать новый параметр для каждого такого подвывода и не использовать его вне подвывода.
        \item Пометка \enquote{Параметр для $N$}: область параметра заканчивается применением правила в строке $N$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Пример 1: перенос существования}
    \begin{itemize}
        \item Докажем $\forall x~(P(x)\to Q(x))\vdash(\exists x~P(x))\to\exists x~Q(x)$.
    \end{itemize}
    {\small
    \begin{align*}
    \begin{nd}[height=3.7ex,topheight=2.5ex]
        \hypo{e1}{\forall x~(P(x)\to Q(x))} \by{Дано}{} \onslide<1->
        \open
            \hypo{e2}{\exists x~P(x)} \by{Допущение для $n$}{} \onslide<2->
            \open[\uncover<3->{a}]
                \hypo{e3}{\text{такой, что }P(a)} \by{Допущение для $n-1$}{} \onslide<3->
                \have{e4}{P(a)\to Q(a)} \byonsubst{6-}{$\forall E$}{e1}{x\mapsto a} \onslide<6->
                \have[n][-3]{e5}{Q(a)} \byon{6-}{$\to E$}{e4,e3} \onslide<5->
                \have[n][-2]{e6}{\exists x~Q(x)} \byonsubst{5-}{$\exists I$}{e5}{x\mapsto a} \onslide<4->
            \close
            \have[n][-1]{e7}{\exists x~Q(x)} \byon{4-}{$\exists E$}{e2,e3-(e6)} \onslide<2->
        \close
        \have[n]{e8}{(\exists x~P(x))\to\exists x~Q(x)} \byon{2-}{$\to I$}{e2-(e7)} \onslide<1->
    \end{nd}
    \end{align*}
    }
\end{frame}

\begin{frame}
    \frametitle{Пример 2: отрицание существования}
    \begin{itemize}
        \item Докажем $\forall y~\neg P(y)\vdash\neg\exists x~P(x)$.
    \end{itemize}
    {\small
    \begin{align*}
    \begin{nd}[height=3.9ex,topheight=2.5ex]
        \hypo{d1}{\forall y~\neg P(y)} \by{Дано}{} \onslide<1->
        \open
            \hypo{d2}{\exists x~P(x)} \by{Допущение для $n$}{} \onslide<2->
            \open[\uncover<3->{a}]
                \hypo{d3}{\text{такой, что }P(a)} \by{Допущение для $n-1$}{} \onslide<3->
                \have{d4}{\neg P(a)} \byonsubst{5-}{$\forall E$}{d1}{y\mapsto a} \onslide<5->
                \have[n][-2]{d5}{\bot} \byon{5-}{$\neg E$}{d4,d3} \onslide<4->
            \close
            \have[n][-1]{d6}{\bot} \byon{4-}{$\exists E$}{d2,d3-(d5)} \onslide<2->
        \close
        \have[n]{d7}{\neg\exists x~P(x)} \byon{2-}{$\neg I$}{d2-(d6)} \onslide<1->
    \end{nd}
    \end{align*}
    }
\end{frame}

\begin{frame}
    \frametitle{Свидетель не является произвольным}
    \setlength{\abovedisplayskip}{2pt}
    \setlength{\belowdisplayskip}{2pt}
    \begin{itemize}
        \item Можно ли вывести $\exists x~P(x)\vdash\forall x~P(x)$?
    \end{itemize}
    {\small
    \begin{align*}
    \begin{nd}[height=3.5ex,topheight=2.5ex]
        \hypo{w1}{\exists x~P(x)} \by{Дано}{} \onslide<1->
        \open[\uncover<2->{a}]
            \hypo{w2}{\text{произв. }a} \by{Параметр для $n$}{} \onslide<2->
            \open[\uncover<3->{b}]
                \hypo{w3}{\text{такой, что }P(b)} \by{Допущение для $n-1$}{} \onslide<3->
                \have[n][-2]{w4}{P(a)} \onslide<3->
            \close
            \have[n][-1]{w5}{P(a)} \byon{3-}{$\exists E$}{w1,w3-(w4)} \onslide<2->
        \close
        \have[n]{w6}{\forall x~P(x)} \byon{2-}{$\forall I$}{w2-(w5)} \onslide<1->
    \end{nd}
    \end{align*}
    }
    \begin{itemize}
        \item<4-> Из $P(b)$ не следует $P(a)$: объекты могут быть разными.
        \item<4-> $a$ уже используется вне подвывода: нужен свежий $b$.
        \item<5-> Контрмодель: $U=\{0,1\}$, $a=1$, $b=0$, $P$ истинно только для $0$.
        Тогда $\exists x~P(x)$ истинно, а $\forall x~P(x)$ ложно.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Пример 3: распределение $\forall$ над $\land$}
    \begin{itemize}
        \item Докажем $\forall x~(P(x)\land Q(x))\vdash(\forall x~P(x))\land(\forall x~Q(x))$.
    \end{itemize}
    {\small
    \begin{align*}
    \begin{nd}[height=3.5ex,topheight=2.2ex]
        \hypo{c1}{\forall x~(P(x)\land Q(x))} \by{Дано}{} \onslide<1->
        \open[\uncover<3->{a_1}]
            \hypo{c2}{\text{произв. }a_1} \by{Параметр для $m$}{} \onslide<3->
            \have{c3}{P(a_1)\land Q(a_1)} \byonsubst{5-}{$\forall E$}{c1}{x\mapsto a_1} \onslide<5->
            \have[m][-1]{c4}{P(a_1)} \byon{5-}{$\land E_l$}{c3} \onslide<4->
        \close
        \have[\alt<1-2>{n-2}{m}]{c5}{\forall x~P(x)} \byon{4-}{$\forall I$}{c2-(c4)} \onslide<2->
        \open[\uncover<3->{a_2}]
            \hypo{c6}{\text{произв. }a_2} \by{Параметр для $n-1$}{} \onslide<3->
            \have{c7}{P(a_2)\land Q(a_2)} \byonsubst{5-}{$\forall E$}{c1}{x\mapsto a_2} \onslide<5->
            \have[n][-2]{c8}{Q(a_2)} \byon{5-}{$\land E_r$}{c7} \onslide<4->
        \close
        \have[n][-1]{c9}{\forall x~Q(x)} \byon{4-}{$\forall I$}{(c6)-(c8)} \onslide<2->
        \have[n]{c10}{(\forall x~P(x))\land(\forall x~Q(x))} \byon{2-}{$\land I$}{c5,c9} \onslide<1->
    \end{nd}
    \end{align*}
    }
\end{frame}

\begin{frame}
    \frametitle{Пример 3 в древовидной записи}
    \begin{itemize}
        \item Докажем $\forall x~(P(x)\land Q(x))\vdash(\forall x~P(x))\land(\forall x~Q(x))$.
    \end{itemize}
    \[
    \begin{prooftree}[separation=1em]
        \hypo{\uncover<3->{\ebsubproof{\text{произв. }a_1}{%
            \begin{prooftree}[center=false]
                \hypo{\uncover<5->{\forall x~(P(x)\land Q(x))\;\checkmark}}
                \ebinfer{5-}{1}{\qrule{\forall E}{x\mapsto a_1}}{\uncover<4->{P(a_1)\land Q(a_1)}}
                \ebinfer{4-}{1}{\land E_l}{P(a_1)}
            \end{prooftree}}}}
        \ebinfer{3-}{1}{\forall I}{\uncover<2->{\forall x~P(x)}}
        \hypo{\uncover<3->{\ebsubproof{\text{произв. }a_2}{%
            \begin{prooftree}[center=false]
                \hypo{\uncover<5->{\forall x~(P(x)\land Q(x))\;\checkmark}}
                \ebinfer{5-}{1}{\qrule{\forall E}{x\mapsto a_2}}{\uncover<4->{P(a_2)\land Q(a_2)}}
                \ebinfer{4-}{1}{\land E_r}{Q(a_2)}
            \end{prooftree}}}}
        \ebinfer{3-}{1}{\forall I}{\uncover<2->{\forall x~Q(x)}}
        \ebinfer{2-}{2}{\land I}{(\forall x~P(x))\land(\forall x~Q(x))}
    \end{prooftree}
    \]
    \begin{itemize}
        \item<5-> Можно ли оба раза использовать $a_1$? \uncover<6->{Да: подвыводы независимы, области параметров не пересекаются.}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Пример 4: перестановка кванторов}
    \begin{itemize}
        \item Докажем $\exists x~\forall y~P(x,y)\vdash\forall y~\exists x~P(x,y)$.
    \end{itemize}
    {\small
    \begin{align*}
    \begin{nd}[height=3.8ex,topheight=2.5ex]
        \hypo{q1}{\exists x~\forall y~P(x,y)} \by{Дано}{} \onslide<1->
        \open[\uncover<2->{a_1}]
            \hypo{q2}{\text{произв. }a_1} \by{Параметр для $n$}{} \onslide<2->
            \open[\uncover<3->{a_2}]
                \hypo{q3}{\text{такой, что }\forall y~P(a_2,y)} \by{Допущение для $n-1$}{} \onslide<3->
                \have{q4}{P(a_2,a_1)} \byonsubst{5-}{$\forall E$}{q3}{y\mapsto a_1} \onslide<4->
                \have[n][-2]{q5}{\exists x~P(x,a_1)} \byonsubst{4-}{$\exists I$}{q4}{x\mapsto a_2} \onslide<3->
            \close
            \have[n][-1]{q6}{\exists x~P(x,a_1)} \byon{3-}{$\exists E$}{q1,q3-(q5)} \onslide<2->
        \close
        \have[n]{q7}{\forall y~\exists x~P(x,y)} \byon{2-}{$\forall I$}{q2-(q6)} \onslide<1->
    \end{nd}
    \end{align*}
    }
\end{frame}

\begin{frame}
    \frametitle{Пример 4 в древовидной записи}
    \begin{itemize}
        \item Докажем $\exists x~\forall y~P(x,y)\vdash\forall y~\exists x~P(x,y)$.
    \end{itemize}
    \[
    \begin{prooftree}
        \hypo{\uncover<2->{\ebsubproof{\text{произв. }a_1}{%
            \begin{prooftree}[center=false]
                \hypo{\uncover<3->{\exists x~\forall y~P(x,y)\;\checkmark}}
                \hypo{\uncover<3->{\ebsubproof{a_2\text{ такой, что }\forall y~P(a_2,y)}{%
                    \begin{prooftree}[center=false]
                        \hypo{\uncover<5->{\forall y~P(a_2,y)\;\checkmark}}
                        \ebinfer{5-}{1}{\qrule{\forall E}{y\mapsto a_1}}{\uncover<4->{P(a_2,a_1)}}
                        \ebinfer{4-}{1}{\qrule{\exists I}{x\mapsto a_2}}{\exists x~P(x,a_1)}
                    \end{prooftree}}}}
                \ebinfer{3-}{2}{\exists E}{\exists x~P(x,a_1)}
            \end{prooftree}}}}
        \ebinfer{2-}{1}{\forall I}{\forall y~\exists x~P(x,y)}
    \end{prooftree}
    \]
    \begin{itemize}
        \item<5-> $a_2$ не входит в результат $\exists E$, а $a_1$ "--- в результат $\forall I$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Построение контрмодели}
    \begin{itemize}
        \item Если поиск вывода зашёл в тупик, ищем контрмодель. Неудача в поиске сама по себе ничего не опровергает. \pause
        \item Выбираем текущую цель. Её доступные посылки и допущения должны быть истинны, а сама цель "--- ложна.
        \item Задаём непустой универсум, значения параметров и интерпретации предикатов на всех наборах объектов. \pause
        \item Иногда нужны дополнительные объекты или бесконечный универсум.
        \item Проверяем найденную модель на исходной формуле или секвенции.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Пример построения контрмодели}
    \begin{itemize}
        \item Проверим $\forall x~(P(x)\lor Q(x))\vdash(\forall x~P(x))\lor(\forall x~Q(x))$. \pause
        \item Пробуем доказать $\forall x~P(x)$: для произвольного $a$ нужна цель $P(a)$, но имеем лишь $P(a)\lor Q(a)$. \pause
        \item Чтобы цель была ложна, а посылка истинна, положим $P(a)=0$, $Q(a)=1$. \pause
        \item Чтобы и $\forall x~Q(x)$ было ложно, добавим объект $b$, где $Q(b)=0$, $P(b)=1$. \pause
        \item Контрмодель: $U=\{0,1\}$, $a=0$, $b=1$.
    \end{itemize}
    \vspace{-2pt}
    \[
        \begin{array}{c|cc}
            x & P(x) & Q(x) \\
            \hline
            0 & 0 & 1 \\
            1 & 1 & 0
        \end{array}
    \]
    \begin{itemize}
        \item Посылка истинна: для каждого объекта истинно $P$ или $Q$.
        Оба дизъюнкта заключения ложны, значит, заключение ложно.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Функциональные символы}
    \begin{itemize}
        \item При наличии функций и констант в $\forall E$ и $\exists I$ используем термы из них и параметров: $f(a_1)$, $a_3+0$.
        \item Как и раньше, без свободных переменных $x,y,\ldots$.
        \item Докажем $\forall x~P(x)\vdash\forall x~P(f(x))$.
    \end{itemize}
    \begin{align*}
    \begin{nd}[height=4ex,topheight=2.5ex]
        \hypo{f1}{\forall x~P(x)} \by{Дано}{} \onslide<1->
        \open[\uncover<2->{a}]
            \hypo{f2}{\text{произв. }a} \by{Параметр для $n$}{} \onslide<2->
            \have[n][-1]{f3}{P(f(a))} \byonsubst{3-}{$\forall E$}{f1}{x\mapsto f(a)} \onslide<2->
        \close
        \have[n]{f4}{\forall x~P(f(x))} \byon{2-}{$\forall I$}{f2-(f3)} \onslide<1->
    \end{nd}
    \end{align*}
\end{frame}

\begin{frame}
    \frametitle{Равенство}
    \begin{itemize}
        \item Правила введения и исключения:
        \[
            \frac{}{t=t}\ =I
            \qquad
            \frac{t=s\quad A[t]}{A[s]}\ =E_l
            \qquad
            \frac{s=t\quad A[t]}{A[s]}\ =E_r.
        \]
        \item $t,s$ "--- термы без свободных переменных. \pause
        \item В $A[t]$ выбираем одно или несколько вхождений $t$. В $A[s]$ заменяем их на $s$.
        \item Замена не должна связывать свободные переменные. \pause
        \item Вместо правил можно использовать рефлексивность и схему аксиом подстановки:
        \[
            \forall x~x=x,\qquad
            \forall x~\forall y~(x=y\land A[x]\to A[y]).
        \]
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Пример: симметричность равенства}
    \begin{itemize}
        \item Докажем $\vdash\forall x~\forall y~(x=y\to y=x)$.
    \end{itemize}
    {\small
    \setlength{\abovedisplayskip}{2pt}
    \setlength{\belowdisplayskip}{2pt}
    \begin{align*}
    \begin{nd}[height=3.4ex,topheight=2.4ex]
        \open[\uncover<2->{a}]
            \hypo{s1}{\text{произв. }a} \by{Параметр для $n$}{} \onslide<2->
            \open[\uncover<3->{b}]
                \hypo{s2}{\text{произв. }b} \by{Параметр для $n-1$}{} \onslide<3->
                \open
                    \hypo{s3}{a=b} \by{Допущение для $n-2$}{} \onslide<4->
                    \have{s4}{\underline{a}=a} \by{$=I$}{} \onslide<5->
                    \have[n][-3]{s5}{b=a} \byonsubst{6-}{$=E_l$}{s3,s4}{a\mapsto b} \onslide<4->
                \close
                \have[n][-2]{s6}{a=b\to b=a} \byon{4-}{$\to I$}{s3-(s5)} \onslide<3->
            \close
            \have[n][-1]{s7}{\forall y~(a=y\to y=a)} \byon{3-}{$\forall I$}{s2-(s6)} \onslide<2->
        \close
        \have[n]{s8}{\forall x~\forall y~(x=y\to y=x)} \byon{2-}{$\forall I$}{s1-(s7)} \onslide<1->
    \end{nd}
    \end{align*}
    }
    \begin{itemize}
        \item<6-> В строке 4 заменяем подчёркнутое $a$ на $b$.
        \item<7-> Получили допустимое правило $\dfrac{t=s}{s=t}\ =\ident{Symm}$.
    \end{itemize}
\end{frame}
