\begin{frame}
\frametitle{Классы моделей}
\begin{itemize}
    \item Практически всегда интересны не все модели данной сигнатуры, а какой-то их класс (или одна модель).
    \item Примеры больших классов моделей: графы, линейные пространства и т.д.
    \item Такие классы часто задаются наборами аксиом.
    \item Натуральные числа (в логике обычно включая $0$) с операциями сложения и умножения "--- \enquote{стандартная модель} для сигнатуры арифметики.
    \item Действительные "--- для мат.\,анализа.
    \item Можно ли задать аксиомы, описывающие конкретную модель с точностью до изоморфизма?
\end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Теории}
    \begin{itemize}
        \item \emph{Теория} $T$ "--- множество замкнутых формул какой-то сигнатуры $\sigma_T$, называемых \emph{аксиомами} $T$. 
        \item Дальше все модели и формулы "--- сигнатуры $\sigma_T$.
        \pause
        \item Модель $M$ называется \emph{моделью $T$}, если в ней истинны все аксиомы $T$. Пишется $M \vDash T$.
        \pause
        \item Замкнутая формула $A$ называется \emph{теоремой $T$} или \emph{выводимой из $T$}, если её можно доказать из аксиом $T$. Пишется $T \vdash A$.
        \pause
        \item Переформулировка теоремы о корректности: $T \vdash A$ $\Rightarrow$ $A$ истинна во всех моделях $T$.
        \item Переформулировка теоремы о полноте: $A$ истинна во всех моделях $T$ $\Rightarrow T \vdash A$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Некоторые простые свойства выводимости}
    \begin{itemize}
        \item Рефлексивность: каждая аксиома $T$ является её теоремой.
        \pause
        \item Транзитивность: если все аксиомы $T_1$ "--- теоремы $T_2$, то все теоремы $T_1$ "--- теоремы $T_2$.
        \pause
        \item Монотонность: если $T_1 \subseteq T_2$, то все теоремы $T_1$ "--- теоремы $T_2$.
        \pause
        \item Теорема о дедукции: если $T \cup \{A\} \vdash B$, то $T \vdash A \to B$.
        \pause
        \item Заметили ли вы, что монотонность следует из рефлексивности и транзитивности?
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Противоречивость теории}
    \begin{itemize}
        \item Теория называется \emph{противоречивой}, если в ней выводимы одновременно какая-то формула $A$ и $\neg A$.
        \item В этом случае в ней выводимы все формулы (её сигнатуры).
        \item Противоречивая теория не имеет моделей.
        \item А непротиворечивая имеет.
        \item Следствие полноты и теоремы Лёвенгейма---Сколема: непротиворечивая теория счётной сигнатуры имеет конечную или счётную модель.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Аксиомы равенства}
    \begin{itemize}
        \item Для доказательства временно рассматриваем $=$ как обычный предикат, без фиксированной интерпретации.
        \item \emph{Нормальная модель} "--- такая, в которой $=$ интерпретируется как равенство объектов, как и раньше.
        \item Какие формулы истинны во всех нормальных моделях?
        \pause
        \item $\forall x ~ x=x$
        \item $\forall x \forall y ~ x=y \to y=x$
        \item $\forall x \forall y \forall z ~ x=y \land y=z \to x=z$
        \item Это рефлексивность, симметричность и транзитивность.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Аксиомы равенства: подстановка}
    \begin{itemize}
        \item Для каждого $n$-местного функционального символа $f$:
        \[
        \begin{gathered}
            \forall x_1\ldots\forall x_n\forall y_1\ldots\forall y_n\\
            x_1=y_1\land\ldots\land x_n=y_n\to\\
            f(x_1,\ldots,x_n)=f(y_1,\ldots,y_n).
        \end{gathered}
        \]
        \pause
        \item Для каждого $n$-местного предикатного символа $P$:
        \[
        \begin{gathered}
            \forall x_1\ldots\forall x_n\forall y_1\ldots\forall y_n\\
            x_1=y_1\land\ldots\land x_n=y_n\to\\
            (P(x_1,\ldots,x_n)\to P(y_1,\ldots,y_n)).
        \end{gathered}
        \]
        \item Вместе с предыдущими тремя это \emph{аксиомы равенства}.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Полнота для нормальных моделей}
    \begin{itemize}
        \item Если в непротиворечивой теории $T$ выводимы все аксиомы равенства её сигнатуры, то $T$ имеет нормальную модель. \pause
        \item Достаточно построить обычную модель $T$, а затем перейти к классам эквивалентности по интерпретации $=$.
        \item Для логики с равенством теоремы Лёвенгейма---Сколема и компактности также верны для нормальных моделей.
        \item Для компактности требуем, чтобы каждое конечное подмножество имело нормальную модель.
        \item Далее аксиомы равенства считаем включёнными во все теории с $=$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Построение нормальной модели}
    \begin{itemize}
        \item По обычной полноте есть модель $M\vDash T$. \pause
        \item Интерпретация $=$ в $M$ задаёт \pause отношение эквивалентности $a\sim b$. \pause
        \item Универсум $N$ состоит из классов эквивалентности $[a]=\{b\mid b\sim a\}$. \pause
        \item $c_N=\pause{}[c_M]$ для каждой константы $c$. \pause
        \item $f_N([a_1],\ldots,[a_n])=\pause{}[f_M(a_1,\ldots,a_n)]$. \pause
        \item $P_N([a_1],\ldots,[a_n])$ истинно тогда и только тогда, когда \pause истинно $P_M(a_1,\ldots,a_n)$. \pause
        \item По аксиомам подстановки результат не зависит от представителей. При этом $=_N$ "--- равенство классов. \pause
        \item Для замкнутой $A$: $M\vDash A\Leftrightarrow N\vDash A$. Поэтому $N\vDash T$. \pause
        \item Далее снова рассматриваем только нормальные модели.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Деревья истинности для теорий}
    \begin{itemize}
        \item Для $T\vdash A$ ищем контрмодель: аксиомам $T$ приписываем $1$, а формуле $A$ "--- $0$.
        \item Аксиомы с $1$ можно добавлять в любую ветвь, не выписывая все в начале.
        \item Если все ветви закрыты, то $T\vdash A$. \pause
        \item Чтобы опровергнуть выводимость, нужна модель \emph{всех} аксиом $T$, в которой $A$ ложна.
        \item Это показывает $T\nvdash A$, но не обязательно $T\vdash\neg A$. \pause
        \item Для $T\vdash\neg A$ начинаем с аксиом и $A=1$ и закрываем дерево.
        \item Конечный вывод использует лишь конечное число аксиом, даже если $T$ бесконечна.
        \item Универсальные аксиомы можно применять повторно, в том числе с новыми параметрами.
        \item Правила для равенства разобраны в семинаре 3.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Пример: частичные порядки}
    \begin{itemize}
        \item Сигнатура: равенство и двухместный предикат $\le$.
        \item Аксиомы (кроме аксиом равенства): \pause
        \begin{enumerate}
            \item Рефлексивность: $\forall x~x\le x$. \pause
            \item Антисимметричность:
            $\forall x\forall y~x\le y\land y\le x\to x=y$. \pause
            \item Транзитивность:
            $\forall x\forall y\forall z~x\le y\land y\le z\to x\le z$.
        \end{enumerate}
        \pause
        \item Модели: $(\mathbb{N},\le)$, $(\mathbb{N},\ge)$, $(\mathbb{Z},\le)$.
        \item Ещё модель: подмножества $\{1,2\}$ (или любого другого множества) с отношением включения.
        \item Добавив $\forall x\forall y~x\le y\lor y\le x$, получим теорию линейных порядков.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Единственность наименьшего элемента}
    \begin{itemize}
        \item Элемент $c$ называется \emph{наименьшим}, если $L(c)$, где $L(t)\coloneqq\forall z~t\le z$.
        \item Теорема: наименьший элемент, если он существует, единственен.
        \[
            \forall x\forall y~L(x)\land L(y)\to x=y.
        \]
        \pause
        \item Пусть $x$ и $y$ "--- наименьшие элементы.
        \item Из $\forall z~x\le z$ при $z\mapsto y$ получаем $x\le y$. \pause
        \item Из $\forall z~y\le z$ при $z\mapsto x$ получаем $y\le x$. \pause
        \item По антисимметричности $x=y$. \pause
        \item Существование наименьшего элемента из аксиом не следует: в $(\mathbb{Z},\le)$ его нет.
    \end{itemize}
\end{frame}

        \item Аксиомы $T$ можно выписывать в любом месте вывода.
    \end{itemize}

\begin{frame}
    \frametitle{Теория графов: смежность}
    \begin{itemize}
        \item Рассмотрим простые неориентированные графы.
        \item Один сорт объектов: вершины.
        \item Предикат $\ident{Adj}(x,y)$: вершины $x$ и $y$ смежны.
        \item Аксиомы (кроме аксиом равенства):
        \begin{enumerate}
            \item $\forall x~\neg \ident{Adj}(x,x)$ \pause
            \item $\forall x\forall y~\ident{Adj}(x,y)\to \ident{Adj}(y,x)$
        \end{enumerate}
        \pause
        \item Петли запрещены, смежность симметрична. Кратные рёбра эта сигнатура не различает.
        \item Что изменится для ориентированных графов?
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Теория графов: инцидентность}
    \begin{itemize}
        \item Два сорта: вершины $V$ и рёбра $E$.
        \item $I(v,e)$: вершина $v$ инцидентна ребру $e$.
        \item Каждое ребро имеет ровно два различных конца:
        \[
        \begin{gathered}
            \forall e:E~\exists v_1:V~\exists v_2:V\\
            I(v_1,e)\land I(v_2,e)\land v_1\ne v_2\land\\
            (\forall v_3:V~I(v_3,e)\to v_3=v_1\lor v_3=v_2).
        \end{gathered}
        \]
        \pause
        \item Два различных конца определяют не более одного ребра:
        \[
        \begin{gathered}
            \forall v_1:V~\forall v_2:V~\forall e_1:E~\forall e_2:E\\
            v_1\ne v_2\land I(v_1,e_1)\land I(v_2,e_1)\land\\
            I(v_1,e_2)\land I(v_2,e_2)\to e_1=e_2.
        \end{gathered}
        \]
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Сравнение двух описаний графов}
    \begin{itemize}
        \item Носители сортов непусты. Поэтому описание через инцидентность требует хотя бы одного ребра.
        \item Описание через смежность допускает и графы без рёбер. \pause
        \item Для графов с рёбрами переведите смежность через инцидентность: \pause
        \[
            \ident{Adj}(v,w)\quad\longleftrightarrow\quad
            v\ne w\land\exists e:E~I(v,e)\land I(w,e).
        \]
        \item Как описать кратные рёбра и петли с помощью инцидентности? \pause
        \item Какая запись удобнее для утверждений о рёбрах? А о вершинах?
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Формальные арифметики}
    \begin{itemize}
        \item Рассмотрим теории первого порядка в языке арифметики. Обычно включают:
        \begin{itemize}
            \item константу $0$;
            \item предикат равенства;
            \item функции $S$ (следующее число), $+$ и $\cdot$.
        \end{itemize}
        \item Обозначения $1,2,\ldots$, $<$ и другие привычные арифметические понятия можно определить через них.
        \item Например, $1=S(0)$.
        \item $x<y$ означает $\exists z~x+S(z)=y$.
        \item Сегодня рассмотрим в первую очередь арифметику Пеано.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Арифметика Пеано}
    \begin{itemize}
        \item $\forall x ~ S(x) \neq 0$
        \item $\forall x \forall y ~ S(x) = S(y) \to x = y$
        \item $\forall x ~ x + 0 = x$
        \item $\forall x \forall y ~ x + S(y) = S(x+y)$
        \item $\forall x ~ x \cdot 0 = 0$
        \item $\forall x \forall y ~ x \cdot S(y) = x \cdot y + x$
        \item Схема индукции: для каждой формулы $A(x,y_1,\ldots,y_n)$ ($x,y_1,\ldots,y_n$ "--- все её свободные переменные): \\
        $\forall y_1 \ldots \forall y_n ~ A(0,y_1,\ldots,y_n) \land \linebreak (\forall x ~ A(x,y_1,\ldots,y_n) \to A(S(x),y_1,\ldots,y_n)) \to \linebreak \forall x ~ A(x,y_1,\ldots,y_n)$
        \pause
        \item Первые шесть аксиом вместе с $\forall x~x\ne0\to\exists y~x=S(y)$ задают \emph{арифметику Робинсона} $Q$ (без индукции).
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Нестандартные модели арифметики}
    \begin{itemize}
        \item Пусть $T$ "--- арифметическая теория, истинная на $\mathbb{N}$ (например, арифметика Пеано). Тогда у неё есть счётная модель, неизоморфная $\mathbb{N}$.
        \item Добавим новую константу $c$ и рассмотрим
        \[
            T'=T\cup\mathit{PA}\cup\{c>0,\ c>1,\ c>2,\ldots\}.
        \]
        \pause
        \item Любое конечное подмножество $T'$ истинно в $\mathbb{N}$, если выбрать достаточно большое значение $c$.
        \item По компактности у $T'$ есть модель, а по теореме Лёвенгейма---Сколема "--- конечная или счётная.
        \item Аксиомы $\mathit{PA}$ исключают конечные модели. Поэтому полученная модель счётна.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Почему модель нестандартна?}
    \begin{itemize}
        \item Пусть $M\vDash T'$. В ней $c>0$, $c>1$, $c>2$, $\ldots$
        \item Забудем интерпретацию $c$: получим модель исходной сигнатуры, удовлетворяющую $T$. \pause
        \item Если она изоморфна $\mathbb{N}$, образ $c_M$ "--- некоторое натуральное число $k$.
        \item Изоморфизм сохраняет порядок и значения числовых термов. Из $c_M>k_M$ получаем $k>k$ в $\mathbb{N}$ "--- противоречие. \pause
        \item Значит, эта счётная модель $T$ неизоморфна $\mathbb{N}$.
        \item Это работает даже для теории всех арифметических истин: их недостаточно для единственности модели.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Полные теории}
    \begin{itemize}
        \item Непротиворечивая теория $T$ называется \emph{полной}, если для каждой замкнутой формулы $A$ (её сигнатуры) либо $T \vdash A$, либо $T \vdash \neg A$.
        \item Для любой модели $M$ теория $\Th(M)$ состоит из всех замкнутых формул, истинных в $M$.
        \item Она непротиворечива: у неё есть модель $M$. \pause
        \item Она полна: если $M\vDash A$, то $\Th(M)\vdash A$, а иначе $\Th(M)\vdash\neg A$.
        \item Полнота не означает единственности модели: даже $\Th(\mathbb{N})$ имеет нестандартные модели.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Пример: плотные линейные порядки}
    \begin{itemize}
        \item Сигнатура: \emph{только порядок} $\le$ и равенство. Без сложения и умножения.
        \item $x<y$ означает $x\le y\land x\ne y$.
        \item К аксиомам линейного порядка добавим:
        \begin{enumerate}
            \item Плотность:
            $\forall x\forall y~x<y\to\exists z~x<z\land z<y$.
            \item Отсутствие наименьшего и наибольшего элементов:
            $\forall x\exists y\exists z~y<x\land x<z$.
        \end{enumerate}
        \pause
        \item Модели: $(\mathbb{Q},\le)$ и $(\mathbb{R},\le)$.
        \item Эта теория \emph{полна} (без доказательства). \pause
        \item Все замкнутые формулы языка порядка имеют одинаковые значения истинности в этих двух моделях.
        \item Но модели неизоморфны: $\mathbb{Q}$ счётно, а $\mathbb{R}$ несчётно.
    \end{itemize}
    \note{Источник: Y. N. Moschovakis, Lecture Notes in Logic, разделы 1F--1G.
    \url{https://www.math.ucla.edu/~ynm/lectures/lnl.pdf}}
\end{frame}

\begin{frame}
    \frametitle{Примеры неполных теорий}
    \begin{itemize}
        \item Полна ли теория графов в сигнатуре смежности? \pause
        \item Нет: $\exists x\exists y~\ident{Adj}(x,y)$ истинно в одних графах и ложно в других.
        \item По корректности ни эта формула, ни её отрицание не выводимы из аксиом графов.
        \pause
        \item Первая теорема Гёделя о неполноте (в усилении Россера): непротиворечивая теория с разрешимым множеством аксиом, в которой выводимы аксиомы арифметики Робинсона, неполна.
        \item В частности, арифметика Пеано неполна.
        \item Подробнее о неполноте в конце курса, если успеем.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Разрешимые теории}
    \begin{itemize}
        \item Теория $T$ называется \emph{разрешимой}, если существует алгоритм, который по замкнутой формуле $A$ определяет, является ли она теоремой $T$.
        \item Для обсуждения алгоритмов далее берём конечные сигнатуры.
        \item В достаточно богатой сигнатуре даже пустая теория неразрешима. \pause
        \item Полная теория с разрешимым множеством аксиом разрешима.
        \item Перебираем все доказательства из $T$, пока не найдём доказательство $A$ или $\neg A$.
        \item Поскольку $T$ полна, одно из них обязательно найдётся.
        \item В первом случае отвечаем \enquote{да}, во втором "--- \enquote{нет}.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Примеры разрешимых теорий}
    \begin{itemize}
        \item $\Th(M)$ для конечной модели $M$ в конечной сигнатуре.
        \item Проверяем формулу в $M$, перебирая значения переменных из конечного универсума. \pause
        \item Теория равенства: сигнатура только $=$, аксиомы равенства.
        \item $\Th(\mathbb{C})$ в сигнатуре $+,\cdot,=$.
        \item Арифметика Пресбургера: арифметика натуральных чисел со сложением, без умножения.
        \item В её схеме индукции тоже используются только формулы языка без умножения.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Примеры неразрешимых теорий}
    \begin{itemize}
        \item Теория частичных порядков.
        \item Теория простых неориентированных графов.
        \item $\Th(\mathbb{N})$ в сигнатуре $+,\cdot,=$.
        \item Арифметика Робинсона и любое её непротиворечивое расширение. \pause
        \item Разрешимость множества аксиом и разрешимость теории "--- разные свойства.
        \item Аксиомы арифметики Пеано распознаются алгоритмом, но алгоритма проверки её теорем нет.
    \end{itemize}
\end{frame}
