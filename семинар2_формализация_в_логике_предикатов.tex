\begin{frame}
    \frametitle{Сигнатуры, термы, формулы: напоминание}
    \begin{itemize}
        \item Сигнатура задаёт константные, функциональные и предикатные символы; у функций и предикатов задана арность.
        \item Термы строятся из переменных и констант применением функций. Они обозначают объекты.
        \item Примеры: $x$, $(y+2)\cdot z$. Замкнутые термы, например $2+2$, не содержат переменных.
        \pause
        \item Атомарные формулы получаются применением предикатов к термам; остальные --- с помощью связок и кванторов.
        \item Без скобок квантор действует на всю формулу справа до конца текущих скобок.
        \item При формализации фиксируем универсум и смысл символов.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Истинность в модели}
    \begin{itemize}
        \item Формула (замкнутая): $\forall x ~ (P(x) \to Q(x))$.
        \item Возьмём модель с носителем $\bar{M} = \{0,1,2\}$ и предикатами \pause
        \item[] \begin{tabular}{c | c c c}
            x & 0 & 1 & 2 \\ \hline
            P(x) & 0 & 0 & 1 \\
            Q(x) & 1 & 0 & 1
        \end{tabular}
        \item Истинна ли формула? \pause По определению: $\sigma_{x \mapsto a}(P(x) \to Q(x)) = 1$ для всех $a : \bar{M}$, то есть
        \[ \begin{array}{l}
            (P(0) \to Q(0)) \land (P(1) \to Q(1)) \land (P(2) \to Q(2)) = \\
            \reveal<+(1)->{(0 \to 1) \land (0 \to 0) \land (1 \to 1) = 1.}
        \end{array} \]
        \item<+(1)-> Переименование объектов $0,1,2$ в $a,b,c$ не изменит ответа.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Истинность в модели (2)}
    \begin{itemize}
        \item Та же модель: $P(x)$ истинно только при $x = 2$, $Q(x)$ "--- при $x = 0$ и $x = 2$.
        \item Формула $(\exists x ~ P(x) \land Q(x)) \land \exists x ~ \neg P(x)$. Истинна ли она? \pause
        \item $\exists x$ "--- \enquote{хотя бы для одного объекта}, то есть дизъюнкция по всем объектам:
        \[ \begin{array}{l}
            \exists x ~ (P(x) \land Q(x)) = \\
            \reveal<+(1)->{(P(0) \land Q(0)) \lor (P(1) \land Q(1)) \lor (P(2) \land Q(2)) =} \\
            \reveal<+(1)->{(0 \land 1) \lor (0 \land 0) \lor (1 \land 1) = 1.}
        \end{array} \]
        \item<.(1)-> Аналогично $\exists x ~ \neg P(x) = \neg P(0) \lor \neg P(1) \lor \neg P(2) = \pause 1 \lor 1 \lor 0 = 1$. \pause
        \item Обе части конъюнкции истинны, значит истинна и вся формула.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Истинность в модели (3)}
    \begin{itemize}
        \item Та же модель:
        \item[] \begin{tabular}{c | c c c}
            x & 0 & 1 & 2 \\ \hline
            P(x) & 0 & 0 & 1 \\
            Q(x) & 1 & 0 & 1
        \end{tabular}
        \item Как насчёт незамкнутой формулы $P(x) \to \forall x ~ Q(x)$? \pause
        \item $\forall x ~ Q(x)$ не зависит от $\sigma(x)$: это $Q(0) \land Q(1) \land Q(2) = \pause 1 \land 0 \land 1 = 0$.
        \item Значит, формула означает $P(x) \to 0$, то есть $\neg P(x)$: \pause истинна при $\sigma(x) = 0,~\sigma(x) = 1$, ложна при $\sigma(x) = 2$.
        \item Для простоты можно писать $x = 0$ вместо $\sigma(x) = 0$. \pause
        \item Значение зависит от $\sigma(x)$, потому что $x$ входит в формулу свободно; вхождение под $\forall x$ связанное и на значение не влияет.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Бинарный предикат}
    \begin{itemize}
        \item Формула: $\forall x \exists y ~ (P(x,y) \to P(x,x))$.
        \item Возьмём $\bar{M} = \{a,b,c\}$.
        \item Какой таблицей задаётся бинарный предикат? \pause
        \item[] \begin{tabular}{c | c c c}
            \diagbox[height=1.5\line]{x}{y} & a & b & c \\ \hline
            a & 0 & 1 & 1 \\
            b & 0 & 1 & 0 \\
            c & 1 & 1 & 0
        \end{tabular}
        \item Истинна ли формула? \pause $\forall x \exists y$: для каждого $x$ хотя бы одно $y$. При $x = a$:
        \begingroup\small % иначе средняя строка не помещается по ширине
        \[ \begin{array}{l}
            \exists y ~ (P(a,y) \to P(a,a)) = \\
            \reveal<+(1)->{(P(a,a) \to P(a,a)) \lor (P(a,b) \to P(a,a)) \lor (P(a,c) \to P(a,a)) =} \\
            \reveal<+(1)->{(0 \to 0) \lor (1 \to 0) \lor (1 \to 0) = 1.}
        \end{array} \]
        \endgroup
        \item<+(1)-> При $x = b$ и $x = c$ аналогично: \pause дизъюнкт с $y = x$ имеет вид $P(x,x) \to P(x,x)$ и истинен всегда. \pause
        \item Это работает в любой модели!
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Бинарный предикат (2)}
    \begin{itemize}
        \item Та же модель:
        \item[] \begin{tabular}{c | c c c}
            \diagbox[height=1.5\line]{x}{y} & a & b & c \\ \hline
            a & 0 & 1 & 1 \\
            b & 0 & 1 & 0 \\
            c & 1 & 1 & 0
        \end{tabular}
        \item Слегка поменяем: $\forall x ~ ((\exists y ~ P(x,y)) \to P(x,x))$. \pause
        \item При $x = a$: $\exists y ~ P(a,y) = P(a,a) \lor P(a,b) \lor P(a,c) = \pause 0 \lor 1 \lor 1 = 1$, а $P(a,a) = 0$, так что импликация даёт \pause $1 \to 0 = 0$.
        \item Для $\forall x$ одного такого $x$ достаточно: формула ложна.
        \pause
        \item На бесконечных моделях полный перебор невозможен; для больших конечных он часто непрактичен.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Формализация}
    \begin{itemize}
        \item Формализация: записать утверждение формулой данной сигнатуры.
        \item Пример: \enquote{Кто-то любит всех на свете}. Универсум: люди, предикат $\ident{Loves}(x, y)$ для \enquote{$x$ любит $y$}.
        \item Кто-то любит всех на свете $\equiv \pause \exists x $ $x$ любит всех на свете $\equiv \pause \exists x ~ \forall y $ $x$ любит $y$ $\equiv \pause \exists x ~ \forall y ~ \ident{Loves}(x, y)$. \pause
        \item А «Каждого кто-то любит»?
        \item Ответ: \pause $\forall y\,\exists x\,\ident{Loves}(x,y)$. Здесь $x$ может зависеть от $y$; в первой формуле один $x$ подходит всем. \pause
        \item Ещё пример в той же сигнатуре: \enquote{Всякая любовь взаимна}.
        \item Ответ: \pause $\forall x ~ \forall y ~ (\ident{Loves}(x, y) \to \ident{Loves}(y,x))$ или \pause $\forall x ~ \forall y ~ (\ident{Loves}(x, y) \leftrightarrow \ident{Loves}(y,x))$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Ещё примеры формализации}
    \begin{itemize}
        \item \enquote{$x$ делится на $2$}. Универсум: натуральные числа.
        \item Если предположили что-то вроде $x / 2 \in \mathbb{N}$: это не подойдёт. Почему? \pause
        \item Например, $\mathbb{N}$ это не объект нашего универсума. А если $\in \mathbb{N}$ рассматривать как единый предикатный символ, он верен для всех объектов! \pause 
        \item Более того, если $/$ "--- функциональный символ, то он должен иметь значение в нашем универсуме для любых аргументов. Есть варианты логики, которые снимают это ограничение, но мы их не изучаем. \pause
        \item Ответ (возможный): \pause $\exists y ~ x = 2 \cdot y$. \pause
        \item Можно ли то же самое записать без $\cdot$? \pause Да! $\exists y ~ x = y + y$. \pause
        \item А \enquote{$x$ делится на $y$} с помощью только $+$, $=$ и числовых констант выразить нельзя (без доказательства).
    \end{itemize}
 \end{frame}



 
\begin{frame}
   \frametitle{Формализация со свободными переменными}
   \begin{itemize}
       \item У нас на предыдущих слайдах появлялись утверждения с переменными (например, \enquote{$x$ любит всех на свете}) в промежуточных результатах. 
       \item Может и сразу быть дано такое утверждение. \pause
       \item Обычно строим формулу с теми же \textbf<3->{свободными} переменными; для связанных выбираем новые имена. \pause
       \item Проверьте свободные переменные: лишняя свободная переменная или случайное связывание исходной часто указывают на ошибку.
   \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Ограниченные кванторы}
    \begin{itemize}
        \item Запись $\forall x>1~x^2>x$ "--- сокращение. Как его раскрыть?
        \pause
        \[ \forall x~(x>1\to x^2>x). \]
        \item А запись $\exists x>1~x^2>x$?
        \pause
        \[ \exists x~(x>1\land x^2>x). \]
        \item<+(1)-> В общем случае:
        \[ \forall x~(R(x)\to A(x)),\qquad \exists x~(R(x)\land A(x)). \]
        \item<.(1)-> Для $\forall$ проверяем $A$ только при выполнении ограничения $R$; для $\exists$ ищем объект, удовлетворяющий обоим условиям.
    \end{itemize}
\end{frame}

\begin{frame}
   \frametitle{\only<1>{$\exists !$}\only<2->{Существование и единственность}}
   \begin{itemize}
       \item $\exists ! x ~ P(x)$ читается как \pause \enquote{Существует единственное $x$ такое, что\ldots}
       \item $P$ здесь "--- формула со свободной переменной $x$. \pause
       \item Но в нашем языке такого символа нет.
       \item Может быть, $!$ "--- предикатный символ (или функциональный, или константный)? \pause Тогда бы не получилась формула. После квантора может стоять только переменная. \pause
       \item Это сокращение, как и ограниченные кванторы. Осталось его расшифровать. \pause
       \item $\exists x ~ (P(x) ~\land~ ???)$ \pause
       \item $\exists x ~ (P(x) \land \forall y ~ (P(y) \to x = y))$ или 
       \item $(\exists x ~ P(x)) \land \forall y,z ~ (P(y) \land P(z) \to y = z)$
   \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Формализация: простые числа}
    \begin{itemize}
        \item Универсум: натуральные числа. Запишем «$x$ --- простое число».
        \item Возможный ответ:
        \pause
        \[ \forall y\,\forall z\,(x=y\cdot z\to y=1\lor z=1). \]
        \item В чём ошибка?
        \pause
        \item Условие выполняется и при $x=1$.
        \pause
        \item Исправим:
        \[ x\ne1\land\forall y\,\forall z\,(x=y\cdot z\to y=1\lor z=1). \]
        \item Если $0$ входит в универсум, он тоже исключается: при $y=z=0$ условие ложно.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Многосортная логика предикатов}
    \begin{itemize}
        \item Иногда нужны разные типы объектов: например, числа, множества чисел и функции.
        \item К сигнатуре добавляются \emph{сорта}. Каждая переменная и константа имеет сорт: $x:S$, $c:S$.
        \item У функций и предикатов заданы сорта аргументов, у функций "--- ещё и сорт результата.
        \item Аргументы должны иметь нужные сорта; иначе запись не является термом или формулой.
        \item В модели каждому сорту соответствует непустой носитель.
        \item Многосортную логику можно свести к односортной: сорта задаются предикатами, а кванторы ограничиваются ими. Нужны также условия на интерпретации символов.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Пример: студенты и задачи}
    \begin{itemize}
        \item Два сорта: $S$ "--- студенты, $T$ "--- задачи.
        \item Каждому студенту назначена ровно одна задача:
        \[ \ident{task}:S\to T. \]
        \item Предикат $\ident{Solved}(s,t)$: студент $s:S$ решил задачу $t:T$.
        \pause
        \item Формализуем: «Каждый студент решил назначенную ему задачу».
        \pause
        \[ \forall s:S\;\ident{Solved}(s,\ident{task}(s)). \]
        \item Здесь $\ident{task}(s)$ имеет сорт $T$, поэтому подходит вторым аргументом $Solved$.
        \pause
        \item При $s:S$, $t:T$ записи $\ident{task}(t)$ и $\ident{Solved}(t,s)$ недопустимы: сорта аргументов не подходят.
        \pause
        \item Упражнение: «Всем студентам назначили разные задачи».
    \end{itemize}
\end{frame}
