\begin{frame}
    \frametitle{Понятие множества}
    \begin{itemize}
        \item Множество "--- это набор (или совокупность) элементов, который рассматривается как единый объект.
        \item \enquote{$a$ "--- элемент $A$} записывается как $a \in A$.
        \item $A$ "--- подмножество $B$ ($A \subseteq B$), если \pause $\forall x ~ (x \in A \to x \in B)$.
        \item $A = B$, если \pause $\forall x ~ (x \in A \leftrightarrow x \in B)$ (или $A \subseteq B \land B \subseteq A$).
        \item Порядок и повторения элементов не важны: $\{a,b,a\}=\{b,a\}$.
        \item Элементом множества может быть другое множество: $\varnothing\neq\{\varnothing\}$.
        \item Не обязательно все элементы множества имеют один \enquote{тип}, т.е. все они числа, или множества, или отрезки$\ldots$
        \item Очень часто говорят о множествах с какой-то дополнительной структурой.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Задание множеств}
    \begin{itemize}
        \item Запись $\{x \mid A(x)\}$ означает совокупность всех $x$, для которых выполняется $A(x)$:
        $\forall x ~ (x \in \{x \mid A(x)\} \leftrightarrow \pause A(x))$.
        \item Не всякая такая совокупность является множеством!
        \item $x$ в $\{x \mid \ldots\}$ "--- связанная переменная.
        \item Задание перечислением: $\{a_1,\ldots,a_n\} = \{x \mid \pause x = a_1 \lor \ldots \lor x = a_n \}$.
        \item Пустое множество: $\varnothing = \{x \mid \pause x \neq x\}$.
        \item Одно множество может быть задано разными эквивалентными свойствами.
        \item Часто значения $x$ берутся из какого-то множества $X$: $\{x : X \mid A(x)\} = \{x \mid x \in X \land A(x)\}$ (или $\{x \in X \mid A(x)\}$).
        \item $x:X$ указывает связанную переменную $x$, пробегающую $X$. Часто вместо $x:X$ пишут $x\in X$.
        \item Слева может быть не переменная, а выражение:
        $\{f(x)\mid x:X\}=\{y\mid\exists x:X~y=f(x)\}$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Операции над множествами}
    \begin{itemize}
        \item Объединение: $A \cup B = \{x \mid \pause x \in A \lor x \in B\}$. \[\bigcup A = \bigcup_{B \in A} B = \reveal<+(1)->{\{x \mid \exists B \in A~x \in B\}}.\]
        \item<.(1)-> Пересечение: $A \cap B = \{x \mid \pause x \in A \land x \in B\} = \{x : A \mid x \in B\}$. Для $A\neq\varnothing$:
        \[\bigcap A = \bigcap_{B \in A} B = \reveal<+(1)->{\{x \mid \forall B \in A~x \in B\}}.\]
        \item<.(1)-> Множество всех подмножеств: $\mathcal{P}(A) = \{ B \mid B \subseteq A \}$ (также обозначается $2^A$).
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Наивная теория множеств и парадокс Рассела}
    \begin{itemize}
        \item Для любого ли свойства есть множество всех объектов, обладающих этим свойством?
        \item \emph{Наивная теория множеств} считала, что да.
        \item Тогда рассмотрим множество всех множеств, не являющихся элементами самих себя: $R = \{X \mid X \notin X\}$.
        \item Вопрос: является ли $R$ элементом самого себя?
        \pause
        \item $R \in R \leftrightarrow R \in \{X \mid X \notin X\} \leftrightarrow \pause R \notin R$.
        \item Пришли к противоречию!
        \pause
        \item Надо как-то ограничить то, какие свойства задают множества.
        \item Для этого задают аксиомы, разрешающие определённые способы построения множеств.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Теории $\ZF$ и $\ZFC$}
    \begin{itemize}
        \item Стандартная аксиоматическая теория множеств "--- \emph{теория Цермело~--- Френкеля} ($\ZF$).
        \item Это теория в логике предикатов. Все объекты "--- множества.
        \item Сигнатура: $\in$ и $=$. Остальные обозначения определяются через них.
        \item $\ZFC$ "--- это $\ZF$ с \emph{аксиомой выбора}.
        \item Аксиома выбора: для любого семейства непустых множеств существует функция, выбирающая по одному элементу из каждого множества семейства.
        \pause
        \item Далее работаем в $\ZFC$. Некоторые утверждения о мощностях используют аксиому выбора.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Аксиомы $\ZF$: неформально}
    \begin{itemize}
        \item \emph{Объёмность:} множества с одними и теми же элементами равны.
        \item \emph{Пустое множество и пара:} существуют $\varnothing$ и $\{a,b\}$.
        \item \emph{Объединение и булеан:} для любого множества $X$ существуют $\bigcup X$ и множество всех подмножеств $\pow{X}$.
        \pause
        \item \emph{Выделение:} для любого множества $X$ и формулы $A$ существует множество $\{x:X\mid A(x)\}$.
        \item \emph{Подстановка:} для любого множества $X$ и формулы $A(x,y)$, если $\forall x:X~\exists!y~A(x,y)$, то существует множество $\{y\mid\exists x:X~A(x,y)\}$.
        \pause
        \item \emph{Бесконечность:} есть множество, содержащее $\varnothing$ и вместе с каждым $x$ содержащее $x\cup\{x\}$.
        \item \emph{Фундирование:} в любом непустом множестве $X$ есть элемент $y$, для которого $y\cap X=\varnothing$.
        \item Формальные формулировки: \href{https://ru.wikipedia.org/wiki/Система_Цермело_—_Френкеля}{Википедия}.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Упорядоченные пары и кортежи}
    \begin{itemize}
        \item \emph{Упорядоченная пара} это способ по любым двум объектам $a$ и $b$ построить такой $(a,b)$, что \pause
        $$(a,b) = (c,d) \leftrightarrow a = c \land b = d$$
        \item Должны быть определимы операции \emph{проекции}: $\ident{pr}_1((a,b))=a$ и $\ident{pr}_2((a,b))=b$.
        \item Это понятие обобщается на упорядоченные тройки, четвёрки и т.д. Общее название "--- \emph{кортеж}.
        \item Тройки можно задать через пары: $(a_1, a_2, a_3) = \pause (a_1, (a_2, a_3))$.
        \item Можно определить пары как множества. Например, определение Куратовского: $$(a, b) = \{\{a\}, \{a, b\}\}$$
        \item Упражнение: задача 6.12 в домашнем задании.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Декартово произведение}
    \begin{itemize}
        \item $A\times B=\{(a,b)\mid a:A,~b:B\}$.
        \item Например, $\{0,1\}\times\{a,b\}=\pause\{(0,a),(0,b),(1,a),(1,b)\}$.
        \pause
        \item Декартово произведение: $$A_1 \times \ldots \times A_n = \reveal<+(1)->{\{(a_1, \ldots, a_n) \mid a_1 : A_1, \ldots, a_n : A_n\}}$$
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Функции}
    \begin{itemize}
        \item \emph{Функция} (или \emph{отображение}) из множества $A$ в множество $B$ сопоставляет каждому элементу $A$ ровно один элемент из $B$:
        $\forall x : A ~ \exists ! y : B ~ y=f(x)$.
        \item $A$ "--- \emph{домен (область определения)} $f$, обозначается $\Dom\ f$.
        \item $B$ "--- \emph{кодомен (область прибытия)} $f$.
        \item \emph{Образ (область значений)} $f$: $\Ran\ f=\{f(x)\mid x:A\}\subseteq B$. Не обязательно совпадает с кодоменом!
        \pause
        \item Если значение функции $f$ на аргументе $x$ задаётся выражением $\ident{expr}$, пишем $f(x) = \ident{expr}$ или $f = x \mapsto \ident{expr}$. 
        \item \emph{Лямбда-выражение} $x \mapsto \ident{expr}$ удобно использовать как часть более сложных выражений.
        \pause
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Функции как множества}
    \begin{itemize}
        \item \emph{График функции} $f:A\to B$ "--- множество пар $\{(x, f(x)) \mid x : A\}$.
        \item У нас функция задаётся тройкой $(A,B,\text{график})$. Есть и другие соглашения.
        \item Можно ли определить $A$ и $B$ по графику? \pause $A$ да, $B$ нет.
        \pause
        \item Множество всех функций из $A$ в $B$ обозначается $A \to B$ или $B^A$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Образы, прообразы}
    \begin{itemize}
        \item Пусть $f:A\to B$. Если $f(x) = y$, то $y$ "--- \emph{образ $x$}, а $x$ "--- \emph{прообраз $y$} (возможно, один из прообразов).
        \item Если $C \subseteq A$, то \emph{образ $C$} это $f[C] = \pause \{f(x) \mid x : C\}$.
        \pause
        \item Если $C \subseteq B$, то \emph{прообраз $C$} это $f^{-1}[C] = \pause \{x : A \mid f(x) \in C\}$.
        \item Обратная функция для этого не нужна.
        \item Часто пишут круглые скобки вместо квадратных, но как раз в теории множеств вполне бывает, что $C \subseteq A$ и $C \in A$ одновременно.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Вложения, наложения и биекции}
    \begin{itemize}
        \item Пусть $f:A\to B$.
        \item $f$ "--- \emph{вложение (инъекция)}, если $\forall x,y:A~x \neq y \to f(x) \neq f(y)$.
        \item То есть каждый элемент $B$ имеет не более одного прообраза.
        \pause
        \item $f$ "--- \emph{наложение (сюръекция)}, если $\forall y:B~\exists x:A~y=f(x)$.
        \item То есть каждый элемент $B$ имеет хотя бы один прообраз. Эквивалентно, $f[A]=B$.
        \pause
        \item $f$ "--- \emph{биекция}, если она одновременно вложение и наложение: $\forall y:B~\exists!x:A~y=f(x)$.
        \item То есть каждый элемент $B$ имеет ровно один прообраз.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Пример: квадрат числа}
    \begin{itemize}
        \item $f:\R\to\R$, $f(x)=x^2$. Домен и кодомен "--- $\R$.
        \item Образ $f[\R]=\pause{}[0,+\infty)$: не совпадает с кодоменом.
        \item $f[\{-2,2\}]=\pause\{4\}$, а $f^{-1}[\{4\}]=\pause\{-2,2\}$.
        \item $f^{-1}[\{-1\}]=\pause\varnothing$.
        \pause
        \item $f$ не инъекция ($f(-2)=f(2)$) и не сюръекция ($-1$ не имеет прообраза).
        \pause
        \item С тем же правилом $x\mapsto x^2$:
        \begin{itemize}
            \item $\R\to[0,+\infty)$ "--- сюръекция, но не инъекция;
            \item $[0,+\infty)\to[0,+\infty)$ "--- биекция.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Мощность множеств}
    \begin{itemize}
        \item Вы уже знакомы с понятием мощности конечных множеств: $|A|$ "--- количество элементов в $A$.
        \item Пусть $A$ и $B$ конечны. В каком случае существует вложение $A$ в $B$ (обозначаем $A \lesssim B$ и говорим \emph{$A$ вкладывается в $B$})? \pause Когда $|A| \leq |B|$.
        \pause
        \item А биекция ($A \sim B$)? \pause Когда $|A| = |B|$.
        \pause
        \item Для любых множеств полагаем $|A|\leq|B|$, если $A\lesssim B$, и $|A|=|B|$, если $A\sim B$.
        \item $A\gtrsim B$ означает $B\lesssim A$, то есть $|A|\geq|B|$.
        \item То есть пока что мощности (кроме натуральных чисел) это какие-то абстрактные объекты, которые можно только сравнивать.
        \item Позже мы увидим, как мощности представить как множества.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Свойства отношений мощности}
    \begin{itemize}
        \item Верно ли, что $\lesssim$ рефлексивно, то есть \pause $\forall A~A \lesssim A$? \pause Да. $\id_A = x \mapsto x$ будет вложением $A$ в $A$.
        \item Транзитивно ли? Пусть $f: A \to B$ и $g: B \to C$ вложения. Можно ли найти вложение $A \to C$? \pause Да, это композиция $g \circ f = x \mapsto \pause g(f(x))$. Доказать, что это действительно вложение $A$ в $C$, несложно.
        \item Симметрично ли? \pause Нет!
        \item Антисимметрично ли? \pause Нет! Например, $\{0\}\lesssim\{1\}$ и $\{1\}\lesssim\{0\}$, но $\{0\}\neq\{1\}$.
        \item Окажется, что не только для конечных: $A \lesssim B \land B \lesssim A \Rightarrow A \sim B$. Это нетривиальная теорема Кантора-Бернштейна, сейчас без доказательства.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Сравнение мощностей и сюръекции}
    \begin{itemize}
        \item В $\ZFC$ любые мощности сравнимы: $A\lesssim B\lor B\lesssim A$.
        \item Это утверждение использует аксиому выбора.
        \pause
        \item Если $B\neq\varnothing$, то $|A|\geq|B|$ тогда и только тогда, когда существует сюръекция $A\to B$.
        \item Для перехода от сюръекции к вложению используем аксиому выбора: выбираем по одному прообразу каждого элемента $B$.
        \item Обратный переход не требует аксиомы выбора.
        \pause
        \item Условие $B\neq\varnothing$ существенно: $|\{0\}|\geq|\varnothing|$, но функции $\{0\}\to\varnothing$ нет.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Свойства отношений мощности}
    \begin{itemize}
        \item А как насчёт $\sim$? Оно тоже рефлексивно: $\id_A$ не только вложение, но и \pause биекция.
        \pause
        \item Транзитивно, потому что \pause композиция биекций окажется биекцией.
        \pause
        \item И симметрично: у любой биекции есть \pause обратная функция, которая тоже окажется биекцией.
        \pause
        \item То есть это отношение \pause эквивалентности.
        \item А $\lesssim$ "--- отношение порядка \enquote{с точностью до $\sim$}.
        \pause
        \item Интуитивно мощность объединяет все равномощные множества в один класс эквивалентности.
        \item Это не определение мощности как множества: для непустого $A$ такой класс слишком велик.
        \item На мощностях $=$ подчиняется аксиомам равенства, а $\leq$ "--- отношение порядка.
        \item Линейность $\leq$ использует аксиому выбора.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Конечные множества}
    \begin{itemize}
        \item Множество $A$ называется \emph{конечным}, если $\exists n \in \N~A \sim \{0,\dots,n-1\}$.
        \item При $n=0$ справа понимается пустое множество.
        \item Соответственно, любое другое множество \emph{бесконечно}.
        \item Это не единственный способ определить конечные и бесконечные множества.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Счётные множества}
    \begin{itemize}
        \item Множество $A$ называется \emph{счётным}, если $A \sim \N$.
        \item Другими словами, все элементы счётного множества можно перенумеровать.
        \item $|\N|$ имеет своё обозначение: $\aleph_0$ (читается \emph{алеф-ноль} или \emph{алеф-нуль}).
        \item Другие счётные множества (докажем на доске): множество чётных чисел, $\mathbb{Z}, \N \times \N$.
        \pause
        \item В этом смысле целых и натуральных чисел \enquote{одинаковое количество}.
        \pause
        \item Любое бесконечное подмножество счётного множества счётно.
        \pause
        \item Всякое бесконечное множество содержит счётное подмножество (используем аксиому выбора).
        \item С аксиомой выбора $\aleph_0$ "--- наименьшая бесконечная мощность: $\aleph_0\leq|A|$ для любого бесконечного $A$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Теорема Кантора}
    \begin{itemize}
        \item А есть ли какие-то бесконечные мощности, кроме $\aleph_0$?
        \pause
        \item Окажется, что для любого $A$ $|A \to \{0,1\}| > |A|$, т.е. $(A \lesssim A \to \{0,1\}) \land (A \nsim A \to \{0,1\})$.
        \item Это называется теоремой Кантора.
        \pause
        \item В том числе $|\N \to \{0,1\}| > |\N|$. То есть множество всех функций из натуральных чисел в $\{0,1\}$ несчётно.
        \item Сначала докажем этот частный случай, затем общий.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Диагональный аргумент для $\N$}
    \begin{itemize}
        \item Допустим, все функции $\N\to\{0,1\}$ перечислены: $f_0,f_1,f_2,\ldots$
        \item Каждая строка "--- значения одной функции:
    \end{itemize}
    \[
    \begin{array}{c|cccc}
           &0&1&2&\cdots\\\hline
        f_0&\alert{f_0(0)}&f_0(1)&f_0(2)&\cdots\\
        f_1&f_1(0)&\alert{f_1(1)}&f_1(2)&\cdots\\
        f_2&f_2(0)&f_2(1)&\alert{f_2(2)}&\cdots\\
        \vdots&\vdots&\vdots&\vdots&\ddots
    \end{array}
    \]
    \begin{itemize}
        \pause
        \item Изменим каждое диагональное значение: $g(n)=1-f_n(n)$.
        \pause
        \item $g\neq f_n$ для любого $n$: они различаются на аргументе $n$.
        \item Значит, $g$ нет в списке. Противоречие!
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Теорема Кантора: общий случай}
    \begin{itemize}
        \item Докажем $|A\to\{0,1\}|>|A|$ для любого множества $A$.
    \end{itemize}
    \begin{enumerate}
        \pause
        \item $A \lesssim A \to \{0,1\}$. Построим вложение
        \[h: A \to (A \to \{0,1\}),\qquad h(x)=\pause(y\mapsto x=y).\]
        Здесь $x=y$ принимает значение $1$, если $x=y$, и $0$ иначе. Разные $x$ дают разные функции $h(x)$.
        \pause
        \item Сюръекции $A\to(A\to\{0,1\})$ нет:
        \begin{itemize}
            \item Допустим, $f$ "--- такая сюръекция. $f(x)$ "--- функция, а $f(x)(x)$ "--- её значение на $x$.
            \item Определим $g:A\to\{0,1\}$: $g(x)=\pause1-f(x)(x)$.
            \item Тогда $g=f(y)$ для некоторого $y\in A$, но
            \[\pause g(y)=1-f(y)(y)\neq f(y)(y)=g(y).\]
            \item Противоречие! В частности, биекции тоже нет.
        \end{itemize}
    \end{enumerate}
\end{frame}

\begin{frame}
    \frametitle{Парадокс Кантора}
    \begin{itemize}
        \item Из теоремы Кантора можно заключить, что не существует множества всех множеств. Почему?
        \pause
        \item Допустим, что оно существует, и обозначим его $U$.
        \item Можно показать, что $\pow{A} \sim A \to \{0,1\}$, поэтому $|\pow{A}| > |A|$ для любого множества.
        \item В том числе $|\pow{U}| > |U|$. Но элементы $\pow{U}$ "--- множества, поэтому \pause $\pow{U} \subseteq U$ и \pause $|\pow{U}| \leq |U|$.
        \item Пришли к противоречию!
        \pause
        \item Заметьте, что функции у нас тоже множества, так что переходить к $\mathcal{P}$ было не обязательно.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Понятие класса}
    \begin{itemize}
        \item Чтобы всё-таки иметь возможность говорить о всех множествах вместе, вводится понятие \emph{класса}.
        \item Это любая совокупность множеств, которые имеют какое-то свойство.
        \item Некоторые классы являются множествами, некоторые нет. 
        \item Вторые называются \emph{собственными классами} и не могут быть элементами классов или множеств.
        \item Ещё один пример собственного класса "--- все одноэлементные множества. Почему?
        \pause
        \item Если бы все одноэлементные множества образовывали множество $S$, то $\bigcup S$ было бы множеством всех множеств.
        \item Но такого множества не существует!
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Классы как объекты: NBG}
    \begin{itemize}
        \item В $\ZFC$ классы "--- способ говорить о свойствах множеств, а не отдельные объекты теории.
        \item В теории фон Неймана~--- Бернайса~--- Гёделя (NBG) все объекты "--- классы.
        \item Множества "--- те классы, которые являются элементами какого-либо класса. Собственные классы элементами быть не могут.
        \item Например, существует класс всех множеств, но он сам не множество.
        \item Формальные формулировки: \href{https://ru.wikipedia.org/wiki/Система_аксиом_фон_Неймана_—_Бернайса_—_Гёделя}{Википедия}.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Операции над мощностями}
    \begin{itemize}
        \item Как можно определить операции над мощностями множеств по аналогии с конечными множествами?
        \item $|A| + |B| = \pause |A \cup B|$, где $A \cap B = \varnothing$. \pause
        \item При необходимости заменяем $A$ и $B$ непересекающимися равномощными копиями $A\times\{0\}$ и $B\times\{1\}$.
        \item $|A| \cdot |B| = \pause |A \times B|$. \pause
        \item $|A|^{|B|} = \pause |B \to A|$.
        \item В частности, $2^{|A|} = \pause |A \to \{0,1\}| = |\pow{A}|$. \pause
        \pause
        \item Важно, что при замене $A$ или $B$ на равномощное результат не меняется. Это нужно доказать (для каждого из определений).
        \item Контрпример: рассмотрим определение $|A| \oplus |B| = |A \cup B|$ (без условий). В чём проблема с ним?
        \item Имеем $\aleph_0 + \aleph_0 = \aleph_0$ и $\aleph_0 \cdot \aleph_0 = \aleph_0$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Мощность континуума}
    \begin{itemize}
        \item $2^{\aleph_0}$ (то есть мощность $\N \to \{0,1\}$ или $\pow{\N}$) имеет своё название \emph{мощность континуума} и обозначение $\continuum$.
        \item Ту же мощность имеют множества $\R$, $(0, 1)$, $\R^n$ при $n\geq1$.
        \item То есть $\continuum + \continuum = \continuum \cdot \continuum = \continuum$.
    \end{itemize}
\end{frame}
