\begin{frame}
\frametitle{Пример дерева истинности}
\begin{itemize}
    \item Начнём с примера. Нужно доказать $p \to (q \to p \land q)$:
    \item[] 
\only<1>{
    \begin{tableau}{
    }
        [{p \to (q \to p \land q) = 0}, just=Дано]
    \end{tableau}
}
\only<2>{
    \begin{tableau}{
        }
        [{p \to (q \to p \land q) = 0}, just=Дано, checked
        [{p = 1}, just=1
        [{q \to p \land q = 0}, just = 1
        ]]]
    \end{tableau}
}
\only<3>{
    \begin{tableau}{
        }
        [{p \to (q \to p \land q) = 0}, just=Дано, checked
        [{p = 1}, just=1, checked
        [{q \to p \land q = 0}, just = 1
        ]]]
    \end{tableau}
}
\only<4>{
    \begin{tableau}{
        }
        [{p \to (q \to p \land q) = 0}, just=Дано, checked
        [{p = 1}, just=1, checked
        [{q \to p \land q = 0}, just = 1, checked
        [{q = 1}, checked, just=3
        [{p \land q = 0}, just=3
        ]]]]]
    \end{tableau}
}
\only<5>{
    \begin{tableau}{
        }
        [{p \to (q \to p \land q) = 0}, just=Дано, checked
        [{p = 1}, just=1, checked
        [{q \to p \land q = 0}, just = 1, checked
        [{q = 1}, checked, just=3
        [{p \land q = 0}, just=3, checked
        [{p = 0}, just=5] [{q = 0}, just=5]
        ]]]]]
    \end{tableau}
}
\only<6>{
    \begin{tableau}{
        }
        [{p \to (q \to p \land q) = 0}, just=Дано, checked
        [{p = 1}, just=1, checked
        [{q \to p \land q = 0}, just = 1, checked
        [{q = 1}, checked, just=3
        [{p \land q = 0}, just=3, checked
        [{p = 0}, just=5, checked, close={2,6}] [{q = 0}, just=5, checked, close={4,6}]
        ]]]]]
    \end{tableau}
}
\onslide<7->{
    \item Все ветви закрыты, поэтому $p \to (q \to p \land q) = 0$ невыполнимо.
    \item Значит, $p \to (q \to p \land q)$ тождественно истинно.
}

\begin{frame}
\frametitle{Следование и порядок применения правил}
\begin{itemize}
    \item Чтобы проверить, следует ли $B$ из посылок $A_1,\, A_2,\,\ldots$, начинаем с $A_1=1,\, A_2=1,\, \ldots,\, B=0$.
    \item Ищем оценку, на которой все посылки истинны, а заключение ложно.
    \pause
    \item Порядок применения правил влияет только на размер дерева, но не на результат.
    \item Обычно выгоднее сначала применять правила без ветвления.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Пример незакрытого дерева}
\begin{itemize}
    \item Проверим $(p \to r) \lor (q \to r) \to (p \lor q \to r)$:
    {\footnotesize
    \begin{tableau}{}
        [{(p \to r) \lor (q \to r) \to (p \lor q \to r) = 0}, checked, just=Дано
        [{(p \to r) \lor (q \to r) = 1}, checked, just=1
        [{p \lor q \to r = 0}, checked, just=1
        [{p \lor q = 1}, checked, just=3
        [{r = 0}, checked, just=3
        [{p \to r = 1}, checked, just=2
        [{p = 1}, checked, just=4
        [{p=0}, checked, just=6, close={7,8}]
        [{r=1}, checked, just=6, close={5,8}]
        ]
        [{q = 1}, checked, just=4
        [\vdots]
        ]
        ]
        [{q \to r = 1}, checked, just=2
        [{p = 1}, checked, just=4
        [{q=0}, checked, just=6]
        [{r=1}, checked, just=6, close={5,8}]
        ]
        [{q = 1}, checked, just=4
        [\vdots]
        ]
        ]
        ]]]]]
    \end{tableau}
    }
    \item В законченной ветви $p=1,\,q=0,\,r=0$. На этом наборе строка 1 выполняется, значит, \pause формула не тождественно истинна.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Ещё один пример}
\begin{itemize}
    \item Следует ли $\lnot r$ из посылок $p \vee (q \vee \lnot r)$, $p \to \lnot r$, $q \to \lnot r$?
    \begin{tableau}{}
        [{p \lor (q \lor \lnot r) = 1}, just=Дано, checked
        [{p \to \lnot r = 1}, just=Дано, checked
        [{q \to \lnot r = 1}, just=Дано, checked,name=last premise
        [{\lnot r = 0}, just=Дано,name=not conc
        [{p = 1}, just={1}, checked
        [{p = 0}, close={:!u,!c}, just={2}, checked]
        [{\lnot r = 1}, close={:not conc,!c},just={2}]]
        [{q \vee \lnot r = 1}, just={1}, checked
        [{q = 1}, move by=1, just={5}, checked
        [{q = 0}, close={:!u,!c}, just={3}, checked]
        [{\lnot r = 1}, close={:not conc,!c},just={3}]]
        [{\lnot r = 1}, close={:not conc,!c},move by=1, just={5}]]]]]]
    \end{tableau}
    \item Дерево закрыто, значит, \pause контрпримера нет и $\lnot r$ действительно следует из посылок.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Деревья с $A$ и $\neg A$}
\begin{itemize}
    \item Вместо формул со знаками часто используют только формулы:
    \[
        A=1 \quad\rightsquigarrow\quad A,
        \qquad
        A=0 \quad\rightsquigarrow\quad \neg A.
    \]
    \item Для двойного отрицания получаем правило $\neg\neg A \Rightarrow A$; остальные правила:
\end{itemize}
\begin{center}
    \begin{tabular}{|c | c | c || c | c | c |}
        \hline
        $\alpha$ & $\alpha_1$ & $\alpha_2$ & $\beta$ & $\beta_1$ & $\beta_2$ \\
        \hline
        $A \land B$ & $A$ & $B$ & $\neg(A \land B)$ & $\neg A$ & $\neg B$ \\
        $\neg(A \lor B)$ & $\neg A$ & $\neg B$ & $A \lor B$ & $A$ & $B$ \\
        $\neg(A \to B)$ & $A$ & $\neg B$ & $A \to B$ & $\neg A$ & $B$ \\
        \hline
    \end{tabular}
\end{center}
\begin{itemize}
    \item Ветвь закрывается, если содержит $A$ и $\neg A$.
    \item Это та же система деревьев: меняется только способ записи.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Пример: импликация как дизъюнкция}
\begin{itemize}
    \item Покажем, что $p \to q$ и $\neg p \lor q$ эквивалентны: проверим следование в обе стороны.
\end{itemize}
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\centering
$p \to q \vDash \neg p \lor q$
{\footnotesize
\begin{tableau}{}
    [{p \to q}, just=Дано, checked
    [{\neg(\neg p \lor q)}, just=Дано, checked
    [{\neg\neg p}, just=2, checked
    [{\neg q}, just=2
    [{p}, just=3, checked
    [{\neg p}, just=1, close={5,6}]
    [{q}, just=1, close={4,6}]
    ]]]]]
\end{tableau}
}
\column{0.49\textwidth}
\centering
$\neg p \lor q \vDash p \to q$
{\footnotesize
\begin{tableau}{}
    [{\neg p \lor q}, just=Дано, checked
    [{\neg(p \to q)}, just=Дано, checked
    [{p}, just=2
    [{\neg q}, just=2
    [{\neg p}, just=1, close={3,5}]
    [{q}, just=1, close={4,5}]
    ]]]]
\end{tableau}
}
\end{columns}
\begin{itemize}
    \item Оба дерева закрыты, поэтому $p \to q$ и $\neg p \lor q$ эквивалентны.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Итоги}
\begin{itemize}
    \item Дерево истинности проверяет совместную выполнимость формул со знаками.
    \item Закрытое дерево показывает, что решений нет; законченная открытая ветвь задаёт решение.
    \item Для проверки тождественной истинности $A$ начинаем с $A=0$: дерево либо доказывает $\vDash A$, либо даёт контрпример.
    \item Построение всегда заканчивается; порядок применения правил влияет только на размер дерева.
    \item Записи $A=1,\,A=0$ и $A,\,\neg A$ задают эквивалентные варианты метода.
\end{itemize}
\end{frame}
